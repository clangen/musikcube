if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT APPLE)
  disable_plugin(projectm)
else()
  set(PROJECTM_ARTIFACT_ROOT "${MUSIKCUBE_VENDOR_ROOT}/bin" CACHE PATH "projectM dependency prefix")
  find_path(PROJECTM_INCLUDE_DIR projectM-4/projectM.h PATHS "${PROJECTM_ARTIFACT_ROOT}/include" ${VENDOR_INCLUDE_DIRECTORIES} NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  find_path(PROJECTM_SDL_INCLUDE_DIR SDL2/SDL.h PATHS "${PROJECTM_ARTIFACT_ROOT}/include" ${VENDOR_INCLUDE_DIRECTORIES} NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  if(APPLE)
    find_package(SDL2 CONFIG QUIET PATHS "${PROJECTM_ARTIFACT_ROOT}/lib/cmake/SDL2" NO_DEFAULT_PATH)
    if(TARGET SDL2::SDL2-static)
      set(PROJECTM_SDL_LIBRARY SDL2::SDL2-static)
    endif()
  else()
    find_path(PROJECTM_X11_INCLUDE_DIR X11/Xlib.h PATHS ${MUSIKCUBE_DEPENDENCY_INCLUDE_DIRS} ${VENDOR_INCLUDE_DIRECTORIES})
    find_library(PROJECTM_SDL_LIBRARY NAMES libSDL2.a PATHS ${VENDOR_LINK_DIRECTORIES} NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  endif()
  find_path(PROJECTM_IMGUI_DIR imgui.cpp PATHS "${PROJECTM_ARTIFACT_ROOT}/include" ${VENDOR_INCLUDE_DIRECTORIES} PATH_SUFFIXES projectm-imgui NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  find_path(PROJECTM_PRESETS_DIR README.md PATHS "${PROJECTM_ARTIFACT_ROOT}/share/projectM" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  if(BUILD_STANDALONE)
    find_library(PROJECTM_LIBRARY NAMES libprojectM-4.a PATHS "${PROJECTM_ARTIFACT_ROOT}/lib" ${VENDOR_LINK_DIRECTORIES} NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
    if(PROJECTM_LIBRARY AND NOT PROJECTM_LIBRARY MATCHES "\\.a$")
      message(FATAL_ERROR "The standalone projectM plugin requires the static projectM library")
    endif()
  else()
    find_library(PROJECTM_LIBRARY NAMES projectM-4 PATHS "${PROJECTM_ARTIFACT_ROOT}/lib" ${VENDOR_LINK_DIRECTORIES})
  endif()
  if(APPLE)
    find_library(PROJECTM_OPENGL_LIBRARY OpenGL)
    set(PROJECTM_X11_INCLUDE_DIR "")
    set(PROJECTM_PLATFORM_READY TRUE)
  else()
    find_library(PROJECTM_OPENGL_LIBRARY NAMES GL PATHS ${VENDOR_LINK_DIRECTORIES})
    set(PROJECTM_PLATFORM_READY ${PROJECTM_X11_INCLUDE_DIR})
  endif()
  if(NOT PROJECTM_INCLUDE_DIR OR NOT PROJECTM_SDL_INCLUDE_DIR OR NOT PROJECTM_PLATFORM_READY OR NOT PROJECTM_IMGUI_DIR OR NOT PROJECTM_PRESETS_DIR OR NOT PROJECTM_SDL_LIBRARY OR NOT PROJECTM_LIBRARY OR NOT PROJECTM_OPENGL_LIBRARY)
    if(BUILD_STANDALONE)
      message(FATAL_ERROR "projectM, presets, static SDL2, Dear ImGui and OpenGL are required for the standalone build")
    endif()
    disable_plugin(projectm)
  else()
    set(PROJECTM_RENDERER_SOURCES
      "${PROJECTM_IMGUI_DIR}/imgui.cpp" "${PROJECTM_IMGUI_DIR}/imgui_draw.cpp"
      "${PROJECTM_IMGUI_DIR}/imgui_tables.cpp" "${PROJECTM_IMGUI_DIR}/imgui_widgets.cpp"
      "${PROJECTM_IMGUI_DIR}/backends/imgui_impl_sdl2.cpp"
      "${PROJECTM_IMGUI_DIR}/backends/imgui_impl_opengl3.cpp")
    if(APPLE)
      add_library(projectm SHARED plugin.cpp macos.cpp)
      add_executable(projectm_helper helper.cpp ${PROJECTM_RENDERER_SOURCES})
      set_target_properties(projectm_helper PROPERTIES OUTPUT_NAME musikcube-projectm
        RUNTIME_OUTPUT_DIRECTORY "${MUSIKCUBE_RUNTIME_OUTPUT_DIRECTORY}/plugins")
      target_compile_features(projectm_helper PRIVATE cxx_std_20)
      target_compile_definitions(projectm_helper PRIVATE SDL_MAIN_HANDLED PROJECTM_STATIC_DEFINE)
      target_include_directories(projectm_helper PRIVATE "${PROJECTM_INCLUDE_DIR}" "${PROJECTM_SDL_INCLUDE_DIR}"
        "${PROJECTM_SDL_INCLUDE_DIR}/SDL2" "${PROJECTM_IMGUI_DIR}")
      target_link_libraries(projectm_helper PRIVATE "${PROJECTM_LIBRARY}" "${PROJECTM_SDL_LIBRARY}"
        "${PROJECTM_OPENGL_LIBRARY}" ${CMAKE_DL_LIBS} Threads::Threads)
      add_dependencies(projectm projectm_helper)
      install(TARGETS projectm_helper RUNTIME DESTINATION share/musikcube/plugins COMPONENT musikcorePlugins)
    else()
      add_library(projectm SHARED plugin.cpp linux.cpp ${PROJECTM_RENDERER_SOURCES})
    endif()
    musikcube_configure_c_plugin(projectm)
    target_compile_features(projectm PRIVATE cxx_std_20)
    target_compile_definitions(projectm PRIVATE PROJECTM_STATIC_DEFINE)
    target_include_directories(projectm PRIVATE "${PROJECTM_INCLUDE_DIR}" "${PROJECTM_SDL_INCLUDE_DIR}"
      "${PROJECTM_SDL_INCLUDE_DIR}/SDL2" "${PROJECTM_IMGUI_DIR}" "${PROJECTM_X11_INCLUDE_DIR}"
      ${MUSIKCUBE_DEPENDENCY_INCLUDE_DIRS})
    if(NOT APPLE)
      target_link_libraries(projectm PRIVATE "${PROJECTM_LIBRARY}" "${PROJECTM_SDL_LIBRARY}" "${PROJECTM_OPENGL_LIBRARY}" ${CMAKE_DL_LIBS} Threads::Threads)
    endif()
    add_custom_command(TARGET projectm POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E remove_directory "$<TARGET_FILE_DIR:projectm>/projectM"
      COMMAND "${CMAKE_COMMAND}" -E copy_directory "${PROJECTM_PRESETS_DIR}" "$<TARGET_FILE_DIR:projectm>/projectM")
    install(DIRECTORY "${PROJECTM_PRESETS_DIR}/" DESTINATION share/musikcube/plugins/projectM COMPONENT musikcorePlugins)
  endif()
endif()
